# PlatformAgent

An agent available to the authenticated user.

## Example Usage

```typescript
import { PlatformAgent } from "@gleanwork/api-client/models/components";

let value: PlatformAgent = {
  name: "HR Policy Agent",
  agent_id: "ef33aecf52c142a6873adf76037b642e",
  capabilities: {},
  subagents: [
    {
      name: "HR Policy Agent",
    },
  ],
};
```
