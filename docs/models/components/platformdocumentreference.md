# PlatformDocumentReference

## Example Usage

```typescript
import { PlatformDocumentReference } from "@gleanwork/api-client/models/components";

let value: PlatformDocumentReference = {
  document_id: "doc_123",
  url: "https://docs.google.com/spreadsheets/d/abc",
  title: "Amoxicillin inventory",
  datasource: "gdrive",
  document_type: "spreadsheet",
  snippets: [
    "Amoxicillin 500mg inventory was last counted on...",
  ],
};
```
