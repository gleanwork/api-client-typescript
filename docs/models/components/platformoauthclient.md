# PlatformOAuthClient

An OAuth client that an admin created for the workspace.

## Example Usage

```typescript
import { PlatformOAuthClient } from "@gleanwork/api-client/models/components";

let value: PlatformOAuthClient = {
  client_id: "<id>",
  display_name: "Russ_Quigley31",
  registration_type: "STATIC",
  grant_types: [
    "TOKEN_EXCHANGE",
  ],
  redirect_uris: [
    "<value 1>",
    "<value 2>",
  ],
  is_enabled: true,
};
```
