# PlatformOAuthClientFilter

A filter on one OAuth client field.

## Example Usage

```typescript
import { PlatformOAuthClientFilter } from "@gleanwork/api-client/models/components";

let value: PlatformOAuthClientFilter = {
  field: "<value>",
  values: [],
};
```
