# PlatformTriggerEvent

A content event a trigger can fire on.

## Example Usage

```typescript
import { PlatformTriggerEvent } from "@gleanwork/api-client/models/components";

let value: PlatformTriggerEvent = {
  event_type: "DOCUMENT_CHANGE",
  datasource: "GITHUB",
  reason: "CREATED",
  doc_type: "Ticket",
  doc_id: "<id>",
  view_url: "https://obvious-publicity.biz",
  title: "<value>",
  event_time: new Date("2024-08-18T23:25:53.054Z"),
};
```
